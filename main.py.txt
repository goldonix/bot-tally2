import os
import logging
from typing import List, Optional
from fastapi import FastAPI, Request
import httpx

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("SocialDistributor")

app = FastAPI(title="Tally Multi-Network Distributor")

# Configuration des clés (qui seront lues depuis Render)
TELEGRAM_BOT_TOKEN = os.getenv("TELEGRAM_BOT_TOKEN", "")
TELEGRAM_CHAT_ID = os.getenv("TELEGRAM_CHAT_ID", "")

FACEBOOK_PAGE_ID = os.getenv("FACEBOOK_PAGE_ID", "")
FACEBOOK_PAGE_ACCESS_TOKEN = os.getenv("FACEBOOK_PAGE_ACCESS_TOKEN", "")

INSTAGRAM_ACCOUNT_ID = os.getenv("INSTAGRAM_ACCOUNT_ID", "")
INSTAGRAM_ACCESS_TOKEN = os.getenv("INSTAGRAM_ACCESS_TOKEN", FACEBOOK_PAGE_ACCESS_TOKEN)

WHATSAPP_PHONE_NUMBER_ID = os.getenv("WHATSAPP_PHONE_NUMBER_ID", "")
WHATSAPP_ACCESS_TOKEN = os.getenv("WHATSAPP_ACCESS_TOKEN", "")
WHATSAPP_RECIPIENT_PHONE = os.getenv("WHATSAPP_RECIPIENT_PHONE", "")
WHATSAPP_TEMPLATE_NAME = os.getenv("WHATSAPP_TEMPLATE_NAME", "hello_world")

async def send_telegram(text: str, image_url: Optional[str] = None):
    async with httpx.AsyncClient() as client:
        if image_url:
            url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendPhoto"
            payload = {"chat_id": TELEGRAM_CHAT_ID, "photo": image_url, "caption": text}
        else:
            url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
            payload = {"chat_id": TELEGRAM_CHAT_ID, "text": text}
        response = await client.post(url, json=payload)
        logger.info(f"Telegram response: {response.status_code}")

async def send_facebook(text: str, image_url: Optional[str] = None):
    async with httpx.AsyncClient() as client:
        if image_url:
            url = f"https://graph.facebook.com/v19.0/{FACEBOOK_PAGE_ID}/photos"
            payload = {"url": image_url, "message": text, "access_token": FACEBOOK_PAGE_ACCESS_TOKEN}
        else:
            url = f"https://graph.facebook.com/v19.0/{FACEBOOK_PAGE_ID}/feed"
            payload = {"message": text, "access_token": FACEBOOK_PAGE_ACCESS_TOKEN}
        response = await client.post(url, data=payload)
        logger.info(f"Facebook response: {response.status_code}")

async def send_instagram(text: str, image_url: str):
    if not image_url:
        return
    async with httpx.AsyncClient() as client:
        create_url = f"https://graph.facebook.com/v19.0/{INSTAGRAM_ACCOUNT_ID}/media"
        create_payload = {"image_url": image_url, "caption": text, "access_token": INSTAGRAM_ACCESS_TOKEN}
        res_container = await client.post(create_url, data=create_payload)
        container_data = res_container.json()
        creation_id = container_data.get("id")
        if not creation_id:
            return
        publish_url = f"https://graph.facebook.com/v19.0/{INSTAGRAM_ACCOUNT_ID}/media_publish"
        publish_payload = {"creation_id": creation_id, "access_token": INSTAGRAM_ACCESS_TOKEN}
        res_publish = await client.post(publish_url, data=publish_payload)
        logger.info(f"Instagram response: {res_publish.status_code}")

async def send_whatsapp(text: str):
    async with httpx.AsyncClient() as client:
        url = f"https://graph.facebook.com/v19.0/{WHATSAPP_PHONE_NUMBER_ID}/messages"
        headers = {"Authorization": f"Bearer {WHATSAPP_ACCESS_TOKEN}"}
        payload = {
            "messaging_product": "whatsapp",
            "to": WHATSAPP_RECIPIENT_PHONE,
            "type": "template",
            "template": {"name": WHATSAPP_TEMPLATE_NAME, "language": {"code": "en_US"}}
        }
        response = await client.post(url, json=payload, headers=headers)
        logger.info(f"WhatsApp response: {response.status_code}")

@app.post("/webhook")
async def handle_tally_webhook(request: Request):
    body = await request.json()
    fields = body.get("data", {}).get("fields", [])
    text_content = ""
    image_url = None
    target_networks: List[str] = []

    for field in fields:
        label = field.get("label", "").lower()
        value = field.get("value")
        if not value:
            continue
        if "texte" in label or "message" in label:
            text_content = str(value)
        elif "réseaux" in label or "site multi" in label or field.get("type") == "MULTIPLE_CHOICE":
            if isinstance(value, list):
                target_networks = [str(v).lower() for v in value]
            elif isinstance(value, str):
                target_networks = [value.lower()]
        elif field.get("type") == "FILE_UPLOAD" and isinstance(value, list) and len(value) > 0:
            image_url = value[0].get("url")

    if any("telegram" in net for net in target_networks):
        await send_telegram(text_content, image_url)
    if any("facebook" in net for net in target_networks):
        await send_facebook(text_content, image_url)
    if any("instagram" in net for net in target_networks):
        await send_instagram(text_content, image_url)
    if any("whatsapp" in net for net in target_networks):
        await send_whatsapp(text_content)

    return {"status": "success", "processed_networks": target_networks}

    port = int(os.environ.get("PORT", 8000))
    uvicorn.run("main:app", host="0.0.0.0", port=port)
